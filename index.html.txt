/* ===== LOOT & EQUIPMENT ===== */
function generateLootDrop(level, guaranteed=false){
 let roll = Math.random(), sum = 0, rar = RARITIES[0];
 for(const r of RARITIES){ sum += r.chance; if(roll <= sum){ rar = r; break; } }
 if(guaranteed) rar = RARITIES[3];
 const types = ["Weapon","Armor","Accessory"], type = types[Math.floor(Math.random()*types.length)];
 const itemLevel = Math.max(1, level + Math.floor(Math.random()*3) - 1);
 const stat = Math.floor((itemLevel * 2 + 6) * rar.mult);
 return {id: Math.random().toString(), name: `${rar.name} Lv.${itemLevel} ${type}`, type, level: itemLevel, rarity: rar.name, stat};
}

function equipmentCompare(item){
 const old = player.equipment[item.type];
 const oldStat = old ? old.stat : 0;
 const diff = item.stat - oldStat;
 return `Current: ${old ? old.name : "None"} (+${oldStat}) | New: ${item.name} (+${item.stat}) | Change: ${diff >= 0 ? "+" : ""}${diff}`;
}

function equipItem(item){
 if(player.equipment[item.type]) player.inventory.push(player.equipment[item.type]);
 player.equipment[item.type] = item;
 player.inventory = player.inventory.filter(i => i.id !== item.id);
 recalculateStats(); updateAll(); toast(`Equipped ${item.name}`);
}

/* ===== STATS & LEVELING ===== */
function recalculateStats(){
 let atk = player.str * 2 + player.dex;
 let def = Math.floor(player.vit / 2);
 let hpBonus = 0;

 for(const type of ["Weapon","Armor","Accessory"]){
  const i = player.equipment[type];
  if(!i) continue;
  if(type === "Weapon") atk += i.stat;
  if(type === "Armor") def += i.stat;
  if(type === "Accessory") hpBonus += i.stat * 5;
 }

 const oldMax = player.maxHp;
 player.maxHp = 100 + player.vit * 10 + hpBonus + Math.max(0, (player.level - 1) * 15);
 player.hp = Math.min(player.maxHp, player.hp + Math.max(0, player.maxHp - oldMax));
 player.atk = atk; player.def = def;
 player.crit = 5 + Math.floor(player.dex / 10) + (player.job === "Archer" ? 10 : 0);
 player.baseSpeed = 3.5 + (player.job === "Rogue" ? 1 : 0) + (mount ? 1.4 : 0);
 player.speed = player.baseSpeed;
}

function addXP(amount){
 if(player.level >= player.maxLevel) return;
 player.xp += amount;
 while(player.xp >= player.xpToNext && player.level < player.maxLevel){
  player.xp -= player.xpToNext;
  player.level++; player.points += 5; player.skillPoints++;
  player.xpToNext = Math.floor(50 * Math.pow(1.22, player.level - 1));
  recalculateStats(); player.hp = player.maxHp;
  toast(`LEVEL UP! Level ${player.level} • +5 Stat Points`);

  if(player.level >= 10) unlockAchievement("level10");
  if(player.level === 10 && !player.baseJob) promptBasicJobSelection();
  if(player.level === 25 && player.baseJob && !player.advancedJob) promptAdvancedJobSelection();
  if(player.level === 40 && player.advancedJob && !player.supremacyJob) promptSupremacyJobSelection();
 }
 updateAll();
}

function spendPoint(stat){
 if(player.points <= 0) return;
 player[stat]++; player.points--; recalculateStats(); updateAll();
}

function makeKit(jobName, tier, baseJob, prefix=""){
 const baseRole = BASE_JOBS[baseJob] ? BASE_JOBS[baseJob].passive : "Mastery";
 let names = ["Strike", "Blast", "Utility", "Ultimate"];

  names = [jobName + " Strike", jobName + " Burst", jobName + " Stance", jobName + " Execution"];
 } else if(tier === 3){
  names = [prefix + " Cut", prefix + " Surge", prefix + " Phase", prefix + " Cataclysm"];
 }
 return {
  passive: `${jobName} ${baseRole}`,
  abilities: [
   ["Q", names[0], "Primary job skill dealing scale ATK damage."],
   ["W", names[1], "Secondary job ability area effect."],
   ["E", names[2], "Mobility, protective shield, or restoration."],
   ["R", names[3], "High-impact ultimate finisher."]
  ]

}

function useSkill(k){
 if(player.cooldowns[k] > 0) return;
 const kit = getCurrentKit();
 const ability = kit.abilities.find(a => a[0] === k);
 if(!ability) return;

 const rank = Math.max(1, player.skillRanks[k] || 0);
 if(k === "R" && (player.skillRanks.R || 0) <= 0){
  toast("Unlock Ultimate with Skill Points first."); return;
 }

 const base = player.baseJob || "Warrior";
 const tier = player.jobTier || 0;
 const scale = 1 + (rank - 1) * 0.18;
 const mult = tier === 3 ? 6.5 : tier === 2 ? 4.5 : 3.0;
 const range = k === "Q" ? 100 : k === "W" ? 160 : k === "E" ? 120 : 250;

 player.cooldowns[k] = Math.max(30, player.maxCooldowns[k] * (base === "Mage" ? 0.82 : 1.0));
 player.isAttacking = true; player.attackTimer = 15;

 if(k === "Q"){
  if(base === "Mage" || base === "Archer"){
   projectiles.push(new Projectile(player.x+16, player.y+20, player.facing === "right" ? 9 : -9, false, player.atk * mult * scale, "#66fcf1", currentZone));
  } else {
   performAreaDamage(range, player.atk * mult * scale, "#66fcf1");
  }
 } else if(k === "W"){
  if(base === "Necromancer"){
   spawnSummon("Skeleton", 2);
  } else {
   performAreaDamage(range, player.atk * mult * 1.2 * scale, "#ab47bc");
  }
 } else if(k === "E"){
  if(base === "Rogue" || base === "Mage"){
   player.x += player.facing === "right" ? 80 : -80;
   player.x = Math.max(0, Math.min(WORLD_WIDTH - player.width, player.x));
  }
  player.hp = Math.min(player.maxHp, player.hp + 40 + player.int * 2 + rank * 10);
  toast("Utility Skill Activated!");
 } else if(k === "R"){
  performAreaDamage(range, player.atk * mult * 2.0 * scale, "#ffd700");
  player.ultimateFlash = {life: 25, color:"#ffd700", radius: range};
  toast(`${ability[1]} Unleashed!`);
 }
 updateAll();
}

function performAreaDamage(range, damage, color){
 mobs.filter(m => m.zone === currentZone && m.alive).forEach(m => {
  const d = Math.hypot(player.x + 16 - (m.x + m.width / 2), player.y + 20 - (m.y + m.height / 2));
  if(d <= range) m.takeDamage(damage);
 });
 spawnParticles(player.x + 16, player.y + 20, color, 18);
}

function upgradeSkill(k){
 if(player.skillPoints <= 0){ toast("No Skill Points available."); return; }
 if(k === "R" && (player.skillRanks.R || 0) < 1){
  player.skillRanks.R = 1; player.skillPoints--; toast("Ultimate Unlocked!"); updateAll(); return;
 }
 if((player.skillRanks[k] || 0) >= 5){ toast("Skill is maxed!"); return; }
 player.skillRanks[k] = (player.skillRanks[k] || 0) + 1;
 player.skillPoints--;
 toast(`${SKILLS[k].name} upgraded to Lv.${player.skillRanks[k]}`);
 updateAll();
}

class Projectile {
 constructor(x,y,vx,isEnemy,damage,color,zone){ Object.assign(this, {x,y,vx,isEnemy,damage,color,zone,alive:true}); }
 update(){
  if(this.zone !== currentZone){ this.alive = false; return; }
  this.x += this.vx;
  if(this.x < 0 || this.x > WORLD_WIDTH){ this.alive = false; return; }
  if(this.isEnemy){
   if(Math.hypot(player.x + 16 - this.x, player.y + 20 - this.y) < 22){
    player.hp = Math.max(0, player.hp - Math.max(1, this.damage - player.def));
    this.alive = false; updateUI();
   }
  } else {
   for(const m of mobs.filter(m => m.zone === currentZone && m.alive)){
    if(Math.hypot(m.x + m.width/2 - this.x, m.y + m.height/2 - this.y) < 22){
     m.takeDamage(this.damage); this.alive = false; break;
    }
   }
  }
 }
 draw(){
  ctx.fillStyle = this.color; ctx.beginPath();
  ctx.arc(this.x - cameraX, this.y, 6, 0, Math.PI * 2); ctx.fill();
 }
}

class Mob {
 constructor(name, type, zone, minX, maxX, y, hp, atk, xp, gold, ranged=false, boss=false, level=1){
  Object.assign(this, {name, type, zone, minX, maxX, y, level, isRanged:ranged, isBoss:boss, alive:true, vx:1.1, respawn:0, shoot:0, anim:0});
  this.x = minX + Math.random() * Math.max(1, maxX - minX - 40);
  this.width = boss ? 48 : 32; this.height = boss ? 48 : 40;
  const scale = 1 + (this.x / WORLD_WIDTH) * .55;
  this.maxHp = Math.floor(hp * scale); this.hp = this.maxHp;
  this.atk = Math.floor(atk * scale); this.xpYield = Math.floor(xp * scale);
  this.goldYield = Math.floor(gold * scale); this.aggro = boss ? 500 : 300;
 }
 update(){
  if(this.zone !== currentZone) return;
  if(!this.alive){ if(--this.respawn <= 0){ this.alive = true; this.hp = this.maxHp; } return; }
  this.anim++;
  const dx = player.x + 16 - (this.x + this.width / 2), dy = player.y + 20 - (this.y + this.height / 2), dist = Math.hypot(dx, dy);
  if(dist < this.aggro){
   this.x += (dx / dist) * (this.isBoss ? 1.0 : 1.45);
   this.y += (dy / dist) * (this.isBoss ? 1.0 : 1.45);
  } else {
   this.x += this.vx;
   if(this.x <= this.minX || this.x >= this.maxX - this.width) this.vx *= -1;
  }
  if(this.isRanged && ++this.shoot > 110){
   this.shoot = 0;
   projectiles.push(new Projectile(this.x + 16, this.y + 16, player.x < this.x ? -5 : 5, true, this.atk, "#ff2d6f", this.zone));
  }
  if(dist < 25) player.hp = Math.max(0, player.hp - this.atk * .025);
 }
 takeDamage(dmg){
  if(!this.alive) return;
  this.hp -= dmg;
  spawnDamage(this.x + 16, this.y, Math.floor(dmg));
  if(this.hp <= 0){
   this.alive = false; this.respawn = 240;
   addXP(this.xpYield); player.gold += this.goldYield;
   unlockAchievement("firstblood");
   if(this.isBoss) unlockAchievement("boss");

   const mat = Math.random();
   if(mat < .28) crafting.iron++;
   else if(mat < .50) crafting.wood++;
   else if(mat < .62) crafting.crystal++;
   else if(mat < .70) crafting.essence++;

   if(Math.random() < .45){
    const item = generateLootDrop(this.level);
    player.inventory.push(item);
    if(item.rarity === "Legendary") unlockAchievement("legendary");
    toast(`Loot: ${item.name}`);
   }
   if(smithQuest.active && !smithQuest.completed && this.zone === 1 && Math.random() < .3){
    smithQuest.current = Math.min(smithQuest.target, smithQuest.current + 1);
    toast(`Goblin Iron ${smithQuest.current}/${smithQuest.target}`);
   }
   const z = ZONES[this.zone];
   if(z.kills !== undefined && z.state === "IN_PROGRESS" && player.jobTier >= z.tier){
    z.kills++; if(z.kills >= z.target) z.state = "READY_TO_TURN_IN";
   }
   if(dungeon.active){
    dungeon.kills++;
    if(dungeon.kills >= 5){
     dungeon.wave++; dungeon.kills = 0;
     if(dungeon.wave > 5){
      dungeon.active = false; player.gold += 500; addXP(1000); unlockAchievement("dungeon");
      toast("Dungeon Cleared! +500G +1000XP");
     } else { toast("Dungeon Wave " + dungeon.wave + "!"); }
    }
   }
   updateAll();
  }
 }
 draw(){
  if(this.zone !== currentZone || !this.alive) return;
  drawSprite(this.x, this.y, this.type, this.anim);
  ctx.fillStyle = "#000a"; ctx.fillRect(this.x - cameraX, this.y - 12, this.width, 4);
  ctx.fillStyle = "#e63946"; ctx.fillRect(this.x - cameraX, this.y - 12, this.width * (this.hp / this.maxHp), 4);
  ctx.fillStyle = "#ffd700"; ctx.font = "9px sans-serif"; ctx.textAlign = "center";
  ctx.fillText(`Lv.${this.level}`, this.x + this.width / 2 - cameraX, this.y - 15);
 }
}

function drawSprite(x, y, type, a=0){
 const sx = x - cameraX;
 ctx.save(); ctx.translate(sx, y);
 ctx.fillStyle = "#0005"; ctx.beginPath(); ctx.ellipse(16, type === "boss" ? 48 : 38, type === "boss" ? 24 : 12, 4, 0, 0, Math.PI * 2); ctx.fill();
 if(type === "player"){
  ctx.fillStyle = "#1d3557"; ctx.fillRect(8,28,6,10); ctx.fillRect(18,28,6,10);
  ctx.fillStyle = "#457b9d"; ctx.fillRect(6,14,20,14); ctx.fillStyle = "#f1faee"; ctx.fillRect(8,2,16,12);
  ctx.fillStyle = "#e63946"; ctx.fillRect(6,0,20,5); ctx.fillStyle = "#111"; ctx.fillRect(player.facing === "right" ? 18 : 10, 6, 3, 3);
  if(player.isAttacking){
   ctx.strokeStyle = "#66fcf1"; ctx.lineWidth = 4; ctx.beginPath();
   ctx.arc(player.facing === "right" ? 28 : 4, 18, 20, player.facing === "right" ? -.9 : 2.2, player.facing === "right" ? .9 : 4.1);
   ctx.stroke();
  }
 } else if(type === "elder"){
  ctx.fillStyle = "#4a148c"; ctx.fillRect(6,10,20,26); ctx.fillStyle = "#ffd54f"; ctx.fillRect(10,10,12,26);
  ctx.fillStyle = "#ffcc80"; ctx.fillRect(10,4,12,8); ctx.fillStyle = "#fff"; ctx.fillRect(8,10,16,12);
 } else if(type === "smith"){
  ctx.fillStyle = "#37474f"; ctx.fillRect(8,14,16,22); ctx.fillStyle = "#bf360c"; ctx.fillRect(8,18,16,14);
  ctx.fillStyle = "#ffcc80"; ctx.fillRect(10,4,12,10);
 } else if(type === "slime"){
  ctx.fillStyle = "#2a9d8f"; ctx.beginPath(); ctx.ellipse(16, 20 + Math.sin(a * .2), 14, 10, 0, 0, Math.PI * 2); ctx.fill();
  ctx.fillStyle = "#fff"; ctx.fillRect(10,16,3,4); ctx.fillRect(18,16,3,4);
 } else if(type === "skeleton"){
  ctx.fillStyle = "#e0e0e0"; ctx.fillRect(12,4,8,8); ctx.fillRect(14,12,4,14);
  ctx.fillRect(8,14,16,3); ctx.fillRect(10,26,4,10); ctx.fillRect(18,26,4,10);
 } else if(type === "drake"){
  ctx.fillStyle = "#d90429"; ctx.fillRect(8,12,16,18); ctx.fillStyle = "#ef233c";
  ctx.beginPath(); ctx.moveTo(8,14); ctx.lineTo(-4, 4 + Math.sin(a * .3) * 6); ctx.lineTo(8,20); ctx.fill();
  ctx.beginPath(); ctx.moveTo(24,14); ctx.lineTo(36, 4 + Math.sin(a * .3) * 6); ctx.lineTo(24,20); ctx.fill();
 } else if(type === "boss"){
  ctx.fillStyle = "#2b2d42"; ctx.fillRect(4,8,40,40); ctx.fillStyle = "#8d99ae";
  ctx.fillRect(10,14,28,28); ctx.fillStyle = "#d90429"; ctx.fillRect(18,20,12,12);
 }
 ctx.restore();
}

 ctx.clearRect(0, 0, canvas.width, 400); drawEnvironment();
 npcs.forEach(n => {
  if(n.zone === currentZone){
   drawSprite(n.x, n.y, n.type, tick);
   ctx.fillStyle = "#66fcf1"; ctx.font = "bold 11px sans-serif"; ctx.textAlign = "center";
   ctx.fillText(n.name, n.x + 16 - cameraX, n.y - 8);
  }
 });
 mobs.forEach(m => m.draw());
 projectiles.forEach(p => p.draw());
 particles.forEach(p => { ctx.fillStyle = p.color; ctx.fillRect(p.x - cameraX, p.y, p.size, p.size); });
 damageNumbers.forEach(d => { ctx.fillStyle = d.color; ctx.font = "bold 12px sans-serif"; ctx.fillText(d.a, d.x - cameraX, d.y); });
 
 player.summons.forEach(s => {
  ctx.fillStyle = s.kind.includes("Skeleton") ? "#cfd8dc" : "#ab47bc";

  ctx.fillStyle = "#fff"; ctx.font = "8px sans-serif"; ctx.textAlign = "center"; ctx.fillText("✦", s.x - cameraX, s.y + 3);
 });

 if(player.ultimateFlash){

  ctx.fillStyle = f.color; ctx.beginPath(); ctx.arc(player.x + 16 - cameraX, player.y + 20, f.radius * (1 - f.life / 25), 0, Math.PI * 2); ctx.fill(); ctx.restore();
 }

 drawSprite(player.x, player.y, "player", player.animFrame || 0);

 const boss = mobs.find(m => m.zone === currentZone && m.alive && m.isBoss && Math.hypot(player.x - m.x, player.y - m.y) < 650);
 const bossbar = document.getElementById("bossbar");
 bossbar.style.display = boss ? "block" : "none";
 if(boss){
  document.getElementById("boss-name").textContent = `${boss.name} • Lv.${boss.level}`;
  document.getElementById("boss-hp").style.width = `${Math.max(0, boss.hp / boss.maxHp * 100)}%`;
 }
}

function update(){
 if(paused) return;
 tick++;
 if(player.attackCooldown > 0) player.attackCooldown--;
 if(player.attackTimer > 0) player.attackTimer--; else player.isAttacking = false;

 let moving = false;
 if(!dialogueNPC && document.getElementById("shop-box").style.display !== "block" && document.getElementById("class-select-box").style.display !== "block"){
  if(keys.ArrowRight || keys.d){ player.x += player.speed; player.facing = "right"; moving = true; }
  if(keys.ArrowLeft || keys.a){ player.x -= player.speed; player.facing = "left"; moving = true; }
  if(keys.ArrowUp || keys.w){ player.y -= player.speed; moving = true; }
  if(keys.ArrowDown || keys.s){ player.y += player.speed; moving = true; }
 }
 player.animFrame = moving ? Math.floor(tick / 8) % 4 : 0;
 player.x = Math.max(0, Math.min(WORLD_WIDTH - player.width, player.x));
 player.y = Math.max(0, Math.min(360, player.y));
 cameraX = Math.max(0, Math.min(WORLD_WIDTH - canvas.width, player.x - canvas.width / 2));

 if(player.x > 2400 && player.y > 140 && player.y < 260) changeZone(1);
 if(player.x < 100 && player.y > 140 && player.y < 260 && ZONES[currentZone].state === "READY_TO_TURN_IN") changeZone(-1);

 Object.keys(player.cooldowns).forEach(k => {
  if(player.cooldowns[k] > 0) player.cooldowns[k]--;
  const el = document.getElementById("cd-" + k.toLowerCase());
  if(player.cooldowns[k] > 0){
   el.style.display = "flex"; el.textContent = Math.ceil(player.cooldowns[k] / 60);
  } else el.style.display = "none";
 });

 mobs.forEach(m => m.update());
 projectiles.forEach(p => p.update());
 for(let i = projectiles.length - 1; i >= 0; i--) if(!projectiles[i].alive) projectiles.splice(i, 1);
 for(let i = particles.length - 1; i >= 0; i--){
  const p = particles[i]; p.x += p.vx; p.y += p.vy; p.life--;
  if(p.life <= 0) particles.splice(i, 1);
 }
 for(let i = damageNumbers.length - 1; i >= 0; i--){
  const d = damageNumbers[i]; d.y -= .5; d.life--;
  if(d.life <= 0) damageNumbers.splice(i, 1);
 }

 player.summons.forEach((s, i) => {
  s.life--;
  const target = mobs.find(m => m.zone === currentZone && m.alive);
  if(target){
   const dx = target.x - s.x, dy = target.y - s.y, d = Math.hypot(dx, dy) || 1;
   s.x += (dx / d) * 1.6; s.y += (dy / d) * 1.6;
   if(d < 30 && tick % 28 === 0) target.takeDamage(s.damage);
  }
  if(s.life <= 0) player.summons.splice(i, 1);
 });

 if(player.ultimateFlash){
  player.ultimateFlash.life--;
  if(player.ultimateFlash.life <= 0) player.ultimateFlash = null;
 }

 if(player.hp <= 0){
  player.hp = player.maxHp; player.x = 100; player.y = 220;
  player.gold = Math.floor(player.gold * .9);
  toast("You were defeated! Lost 10% gold.");
 }
}
