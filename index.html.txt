<script>
const canvas = document.getElementById("gameCanvas");
const ctx = canvas.getContext("2d");
const WORLD_WIDTH = 2500;

let cameraX = 0, tick = 0, paused = false;
let windowToastTimer = null;
const keys = {};
const panelIds = ["stat-window","inventory-window","quest-window","skill-window","shop-box","class-select-box","character-box","pause-box","rpg-panel"];

const BASE_JOBS = {
 Warrior:{str:15,dex:8,int:4,vit:14,passive:"Rage Engine",lore:"Frontline melee champions."},
 Rogue:{str:8,dex:18,int:6,vit:8,passive:"Killing Intent",lore:"Elusive assassin specializing in critical strikes."},
 Tank:{str:10,dex:6,int:4,vit:20,passive:"Unbreakable",lore:"Living fortresses built to absorb damage."},
 Mage:{str:4,dex:8,int:20,vit:6,passive:"Arcane Mastery",lore:"Manipulates raw area magic."}
};

const BASE_KITS = {
 Warrior:{passive:"Rage Engine",abilities:[["Q","Cleave","Melee strike."],["W","War Cry","Bonus ATK."],["E","Second Wind","Heal."],["R","Titan Cleave","Shockwave."]]},
 Rogue:{passive:"Killing Intent",abilities:[["Q","Shadowstep","Blink forward."],["W","Venom Fan","Poison attack."],["E","Vanish","Speed boost."],["R","Assassinate","Lethal burst."]]},
 Tank:{passive:"Unbreakable",abilities:[["Q","Shield Bash","Stun strike."],["W","Fortify","Armor boost."],["E","Taunt","Weaken foes."],["R","Aegis","Defensive barrier."]]},
 Mage:{passive:"Arcane Mastery",abilities:[["Q","Arcane Bolt","Magic projectile."],["W","Meteor","Area blast."],["E","Blink","Teleport."],["R","Sunfall","Solar drop."]]}
};

const ZONES = [
 {name:"Oakhaven Town",req:"Talk to Elder Eldrin",tier:0,state:"NOT_STARTED"},
 {name:"Whispering Swamp",req:"Defeat 8 Slimes",tier:1,kills:0,target:8,state:"LOCKED"},
 {name:"Ashen Caverns",req:"Defeat 10 Skeletons",tier:2,kills:0,target:10,state:"LOCKED"},
 {name:"Dragon's Peak",req:"Defeat 3 VOID TITANS",tier:3,kills:0,target:3,state:"LOCKED"}
];

const player = {
 name:"Hero", x:100, y:220, width:32, height:40, speed:3.5,
 hp:100, maxHp:100, level:1, xp:0, xpToNext:50, maxLevel:50, gold:0,
 job:"Novice", jobTier:0, baseJob:"",
 str:5, dex:5, int:5, vit:5, points:0, atk:12, def:2, crit:5,
 attackCooldown:0, attackTimer:0, isAttacking:false, facing:"right",
 cooldowns:{Q:0,W:0,E:0,R:0}, maxCooldowns:{Q:120,W:240,E:360,R:720},
 skillRanks:{Q:1,W:1,E:1,R:0}, skillPoints:0,
 inventory:[], equipment:{Weapon:null,Armor:null,Accessory:null}
};

let currentZone = 0, dialogueNPC = null, dialogueLines = [], dialogueIndex = 0;
const projectiles = [], particles = [], damageNumbers = [];

function toast(msg){
 const el = document.getElementById("toast");
 el.textContent = msg; el.style.display = "block";
 clearTimeout(windowToastTimer);
 windowToastTimer = setTimeout(()=> el.style.display="none", 2200);
}

function closePanels(){
 panelIds.forEach(id => document.getElementById(id).style.display = "none");
 dialogueNPC = null;
 document.getElementById("dialogue-box").style.display = "none";
}

function openPanel(id){
 closePanels();
 document.getElementById(id).style.display = "block";
}

function recalculateStats(){
 let atk = player.str * 2 + player.dex;
 let def = Math.floor(player.vit / 2);
 if(player.equipment.Weapon) atk += player.equipment.Weapon.stat;
 if(player.equipment.Armor) def += player.equipment.Armor.stat;

 player.maxHp = 100 + player.vit * 10;
 player.hp = Math.min(player.maxHp, player.hp);
 player.atk = atk; player.def = def;
}

function addXP(amount){
 player.xp += amount;
 while(player.xp >= player.xpToNext && player.level < player.maxLevel){
  player.xp -= player.xpToNext;
  player.level++; player.points += 5; player.skillPoints++;
  player.xpToNext = Math.floor(50 * Math.pow(1.2, player.level - 1));
  recalculateStats(); player.hp = player.maxHp;
  toast(`LEVEL UP! Level ${player.level}`);
 }
 updateUI();
}

function spendPoint(stat){
 if(player.points <= 0) return;
 player[stat]++; player.points--; recalculateStats(); updateUI();
}

function getCurrentKit(){
 return BASE_KITS[player.baseJob] || BASE_KITS.Warrior;
}

function useSkill(k){
 if(player.cooldowns[k] > 0) return;
 player.cooldowns[k] = player.maxCooldowns[k];
 player.isAttacking = true; player.attackTimer = 15;
 toast(`${k} Ability Used!`);
}

function upgradeSkill(k){
 if(player.skillPoints <= 0) return;
 player.skillRanks[k] = (player.skillRanks[k] || 0) + 1;
 player.skillPoints--; updateUI();
}

function selectJob(name){
 player.baseJob = name; player.job = name; player.jobTier = 1;
 const s = BASE_JOBS[name]; player.str = s.str; player.dex = s.dex; player.int = s.int; player.vit = s.vit;
 recalculateStats(); closePanels(); updateUI(); toast(`${name} Class Selected!`);
}

function promptBasicJobSelection(){
 openPanel("class-select-box");
 const o = document.getElementById("class-options"); o.innerHTML = "";
 for(const [name, s] of Object.entries(BASE_JOBS)){
  const b = document.createElement("button"); b.style.width = "100%"; b.style.margin = "4px 0";
  b.innerHTML = `<b>${name}</b><br><small>${s.lore}</small>`;
  b.onclick = () => selectJob(name);
  o.appendChild(b);
 }
}

function melee(){
 if(player.attackCooldown > 0) return;
 player.attackCooldown = 18; player.isAttacking = true; player.attackTimer = 12;
 mobs.filter(m => m.zone === currentZone && m.alive).forEach(m => {
  if(Math.hypot(player.x - m.x, player.y - m.y) < 50) m.takeDamage(player.atk);
 });
}

class Mob {
 constructor(name, type, zone, x, y, hp, atk, xp, gold){
  Object.assign(this, {name, type, zone, x, y, maxHp:hp, hp, atk, xpYield:xp, goldYield:gold, alive:true, width:32, height:40});
 }
 update(){
  if(this.zone !== currentZone || !this.alive) return;
  const dx = player.x - this.x, dy = player.y - this.y, dist = Math.hypot(dx, dy);
  if(dist < 200){ this.x += (dx / dist) * 1.2; this.y += (dy / dist) * 1.2; }
  if(dist < 25) player.hp = Math.max(0, player.hp - this.atk * .02);
 }
 takeDamage(dmg){
  this.hp -= dmg;
  if(this.hp <= 0){
   this.alive = false; addXP(this.xpYield); player.gold += this.goldYield;
   const z = ZONES[this.zone];
   if(z.kills !== undefined){ z.kills++; if(z.kills >= z.target) z.state = "COMPLETED"; }
   updateUI();
  }
 }
 draw(){
  if(this.zone !== currentZone || !this.alive) return;
  ctx.fillStyle = "#e63946"; ctx.fillRect(this.x - cameraX, this.y, 32, 40);
 }
}

const mobs = [
 new Mob("Slime","slime",1,300,200,60,8,20,15),
 new Mob("Skeleton","skeleton",2,400,220,150,18,50,40)
];

const npcs = [{name:"Elder Eldrin",zone:0,x:180,y:200}];

function interact(){
 const npc = npcs.find(n => n.zone === currentZone && Math.hypot(player.x - n.x, player.y - n.y) < 60);
 if(!npc) return;
 dialogueNPC = npc;
 dialogueLines = ["Welcome to Oakhaven!", "Select your job class when ready."];
 document.getElementById("dialogue-name").textContent = npc.name;
 document.getElementById("dialogue-text").textContent = dialogueLines[0];
 document.getElementById("dialogue-box").style.display = "block";
 promptBasicJobSelection();
}

function advanceDialogue(){ closePanels(); }

function applyCharacter(){
 player.name = document.getElementById("char-name").value || "Hero";
 closePanels(); updateUI();
}

function togglePause(){ paused = !paused; document.getElementById("pause-box").style.display = paused ? "block" : "none"; }
function resetGame(){ location.reload(); }

function updateUI(){
 document.getElementById("zone-display").textContent = ZONES[currentZone].name;
 document.getElementById("job-display").textContent = player.job;
 document.getElementById("level-display").textContent = player.level;
 document.getElementById("hp-display").textContent = `${Math.ceil(player.hp)}/${player.maxHp}`;
 document.getElementById("xp-display").textContent = `${player.xp}/${player.xpToNext}`;
 document.getElementById("gold-display").textContent = player.gold;
 document.getElementById("st-str").textContent = player.str;
 document.getElementById("st-dex").textContent = player.dex;
 document.getElementById("st-int").textContent = player.int;
 document.getElementById("st-vit").textContent = player.vit;
 document.getElementById("st-points").textContent = player.points;
 document.getElementById("st-atk").textContent = player.atk;
 document.getElementById("st-def").textContent = player.def;
}

function update(){
 if(paused) return;
 tick++;
 if(player.attackCooldown > 0) player.attackCooldown--;

 if(keys.ArrowRight || keys.d){ player.x += player.speed; player.facing = "right"; }
 if(keys.ArrowLeft || keys.a){ player.x -= player.speed; player.facing = "left"; }
 if(keys.ArrowUp || keys.w){ player.y -= player.speed; }
 if(keys.ArrowDown || keys.s){ player.y += player.speed; }

 player.x = Math.max(0, Math.min(WORLD_WIDTH - player.width, player.x));
 player.y = Math.max(0, Math.min(360, player.y));
 cameraX = Math.max(0, Math.min(WORLD_WIDTH - canvas.width, player.x - canvas.width / 2));

 mobs.forEach(m => m.update());
}

function draw(){
 ctx.clearRect(0, 0, canvas.width, 400);
 ctx.fillStyle = "#20252b"; ctx.fillRect(0,0,canvas.width,400);
 npcs.forEach(n => {
  if(n.zone === currentZone){
   ctx.fillStyle = "#ffd54f"; ctx.fillRect(n.x - cameraX, n.y, 32, 40);
  }
 });
 mobs.forEach(m => m.draw());
 ctx.fillStyle = "#66fcf1"; ctx.fillRect(player.x - cameraX, player.y, 32, 40);
}

window.addEventListener("keydown", e => {
 keys[e.key] = true;
 const k = e.key.toLowerCase();
 if(k === "j") melee();
 if(["q","w","e","r"].includes(k)) useSkill(k.toUpperCase());
 if(k === "f" || e.key === " ") interact();
 if(k === "c") openPanel("stat-window");
 if(k === "i") openPanel("inventory-window");
 if(k === "escape") togglePause();
});

window.addEventListener("keyup", e => keys[e.key] = false);

// GAME LOOP ENGINE SETUP
function loop(){
 update();
 draw();
 requestAnimationFrame(loop);
}

// INITIALIZATION
recalculateStats();
updateUI();
requestAnimationFrame(loop);
</script>
